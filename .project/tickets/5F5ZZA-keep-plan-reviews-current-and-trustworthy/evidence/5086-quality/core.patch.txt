+const packageRoot = nodePath.resolve(import.meta.dirname, '..');
+const evidenceDirectory = nodePath.join(packageRoot, 'tests/fixtures/reviewer-capability-evidence');
+const cataloguePath = nodePath.join(packageRoot, 'src/review/capability-catalogue.generated.ts');
+const rubrics = REVIEWER_CAPABILITY_RUBRICS;

+function modelFor(reviewer: 'claude' | 'codex'): string {
+  const key = reviewer === 'claude' ? 'anthropic_model' : 'openai_model';
+  const model = REVIEWER_CAPABILITY_MANIFEST.settings[key];
+  if (model === undefined || model === '') throw new Error(`Missing pinned ${key}`);
+  return model;
+}
+
+function evidencePath(reviewer: 'claude' | 'codex'): string {
+  return nodePath.join(evidenceDirectory, `${modelFor(reviewer)}.json`);
+}
+
+function readEvidence(reviewer: 'claude' | 'codex'): SealedCapabilityResults {
+  return JSON.parse(readFileSync(evidencePath(reviewer), 'utf8')) as SealedCapabilityResults;
+}
+
+function passingRuns(result: SealedCapabilityResults): number {
+  const labels = new Map(
+    REVIEWER_CAPABILITY_MANIFEST.fixtures.map(fixture => [fixture.label.id, fixture.label]),
+  );
+  return result.runs.filter(run => {
+    const label = labels.get(run.fixture_id);
+    return label !== undefined && scoreCapabilityRun(label, run);
+  }).length;
+}

+async function record(reviewer: 'claude' | 'codex'): Promise<void> {
+  process.env.SAFEWORD_REVIEW_WORKER = '1';
+  process.env.SAFEWORD_REVIEW_TIMEOUT_MS = REVIEWER_CAPABILITY_MANIFEST.settings.review_timeout_ms;
+  process.env.SAFEWORD_REVIEW_RUN_BOUND_MS =
+    REVIEWER_CAPABILITY_MANIFEST.settings.review_run_bound_ms;
+  const provider = reviewer === 'claude' ? 'anthropic' : 'openai';
+  const model = modelFor(reviewer);
+  let attempted = 0;
+  const result = await collectCapabilityResults({
+    manifest: REVIEWER_CAPABILITY_MANIFEST,
+    rubrics,
+    provider,
+    model,
+    evidenceDate: new Date().toISOString().slice(0, 10),
+    review: packet => {
+      attempted += 1;
+      process.stdout.write(`${attempted}: ${packet.dispatch_id}\n`);
+      return runHeadlessReviewerWithProvenance(reviewer, packet, packageRoot, packageRoot, {
+        model,
+        effort: 'medium',
+      });
+    },
+  });
+  mkdirSync(evidenceDirectory, { recursive: true });
+  writeFileSync(evidencePath(reviewer), `${JSON.stringify(result, undefined, 2)}\n`);
+  const passes = passingRuns(result);
+  process.stdout.write(`${model}: ${passes}/${result.runs.length} scored runs passed\n`);
+  if (
+    compareSealedCapabilityResults(REVIEWER_CAPABILITY_MANIFEST, rubrics, result, result) !==
+    'not_weaker'
+  ) {
+    process.exitCode = 1;
+  }
+}

+async function catalogueSource(): Promise<string> {
+  const claude = readEvidence('claude');
+  const codex = readEvidence('codex');
+  if (
+    claude.provider !== 'anthropic' ||
+    claude.model !== modelFor('claude') ||
+    codex.provider !== 'openai' ||
+    codex.model !== modelFor('codex')
+  ) {
+    throw new Error('Capability evidence does not match the pinned exact models');
+  }
+  const forward = compareSealedCapabilityResults(
+    REVIEWER_CAPABILITY_MANIFEST,
+    rubrics,
+    claude,
+    codex,
+  );
+  const reverse = compareSealedCapabilityResults(
+    REVIEWER_CAPABILITY_MANIFEST,
+    rubrics,
+    codex,
+    claude,
+  );
+  process.stdout.write(`Claude to Codex: ${forward}; Codex to Claude: ${reverse}\n`);
+  if (forward === 'unknown' || reverse === 'unknown') {
+    throw new Error('Shipped default cross-provider capability ordering is unqualified');
+  }
+  const revision = {
+    corpus_digest: claude.corpus_digest,
+    rubric_digest: claude.rubric_digest,
+    settings_digest: claude.settings_digest,
+  };
+  const pairs = (
+    [
+      [claude, codex, forward],
+      [codex, claude, reverse],
+    ] as const
+  ).map(([author, reviewer, direction]) => ({
+    ...revision,
+    author_provider: author.provider,
+    author_model: author.model,
+    reviewer_provider: reviewer.provider,
+    reviewer_model: reviewer.model,
+    direction,
+    qualification: 'pinned-corpus',
+    evidence_date:
+      author.evidence_date.localeCompare(reviewer.evidence_date) > 0
+        ? author.evidence_date
+        : reviewer.evidence_date,
+    results_digest: createHash('sha256')
+      .update(`${author.results_digest}\0${reviewer.results_digest}`)
+      .digest('hex'),
+  }));
+  const source = `// Generated by scripts/reviewer-capability-eval.ts from sealed live evidence.\nexport const PACKAGED_CAPABILITY_REVISION = ${JSON.stringify(revision, undefined, 2)} as const;\n\nexport const PACKAGED_CAPABILITY_SETTINGS = ${JSON.stringify(REVIEWER_CAPABILITY_MANIFEST.settings, undefined, 2)} as const;\n\nexport const PACKAGED_CAPABILITY_PAIRS = ${JSON.stringify(pairs, undefined, 2)} as const;\n`;
+  const options = await prettier.resolveConfig(cataloguePath);
+  return prettier.format(source, { ...options, filepath: cataloguePath });
+}

+async function verify(): Promise<void> {
+  if (readFileSync(cataloguePath, 'utf8') !== (await catalogueSource())) {
+    throw new Error('Packaged capability catalogue is stale; regenerate from current evidence');
+  }
+}

+const [command, reviewer] = process.argv.slice(2);
+try {
+  if (command === undefined || command === 'verify') await verify();
+  else if (command === 'generate') writeFileSync(cataloguePath, await catalogueSource());
+  else if (command === 'record' && (reviewer === 'claude' || reviewer === 'codex')) {
+    await record(reviewer);
+  } else if (command === 'rescore' && reviewer !== undefined) {
+    const previous = JSON.parse(readFileSync(reviewer, 'utf8')) as CapabilityManifest;
+    // Compute both before writing either: a failed migration leaves both original files intact.
+    const rescored = (['claude', 'codex'] as const).map(agent => ({
+      agent,
+      result: rescoreCapabilityResults(
+        previous,
+        REVIEWER_CAPABILITY_MANIFEST,
+        rubrics,
+        readEvidence(agent),
+        new Date().toISOString(),
+      ),
+    }));
+    for (const { agent, result } of rescored) {
+      writeFileSync(evidencePath(agent), `${JSON.stringify(result, undefined, 2)}\n`);
+      process.stdout.write(
+        `${result.model}: ${passingRuns(result)}/${result.runs.length} rescored observations; collected ${result.evidence_date}; original seal ${result.rescored_from?.results_digest}\n`,
+      );
+    }
+  } else {
+    throw new Error(
+      'Usage: bun run test:eval:reviewer-capability [verify | generate | record claude | record codex | rescore <previous-manifest.json>]',
+    );
+  }
+} catch (error) {
+  process.stderr.write(`${String(error)}\n`);

+import {
+  PACKAGED_CAPABILITY_PAIRS as generatedPairs,
+  PACKAGED_CAPABILITY_REVISION as generatedRevision,
+} from './capability-catalogue.generated.js';
+
+export interface CapabilityRevision {
+  readonly corpus_digest: string;
+  readonly rubric_digest: string;
+  readonly settings_digest: string;
+}
+
+export interface CapabilityPairRecord extends CapabilityRevision {
+  readonly author_provider: string;
+  readonly reviewer_provider: string;
+  readonly author_model: string;
+  readonly reviewer_model: string;
+  readonly direction: 'not_weaker' | 'weaker';
+  readonly qualification: 'pinned-corpus' | 'provider-order' | 'unqualified';
+  readonly evidence_date: string;
+  readonly results_digest: string;
+}
+
+export type CapabilityComparison = 'not_weaker' | 'weaker' | 'unknown';
+
+export interface CapabilityModelIdentity {
+  readonly provider: string;
+  readonly model: string;
+}
+
+export const PACKAGED_CAPABILITY_REVISION: CapabilityRevision = generatedRevision;
+export const PACKAGED_CAPABILITY_PAIRS: readonly CapabilityPairRecord[] = generatedPairs;
+
+function matchesRevision(record: CapabilityPairRecord, current: CapabilityRevision): boolean {
+  return (
+    record.corpus_digest === current.corpus_digest &&
+    record.rubric_digest === current.rubric_digest &&
+    record.settings_digest === current.settings_digest
+  );
+}
+
+function hasQualifiedEvidence(record: CapabilityPairRecord): boolean {
+  return (
+    record.qualification !== 'unqualified' &&
+    record.author_provider !== '' &&
+    record.reviewer_provider !== '' &&
+    (record.qualification !== 'provider-order' ||
+      record.author_provider === record.reviewer_provider) &&
+    record.evidence_date !== '' &&
+    record.results_digest !== ''
+  );
+}
+
+/** Only current, qualified evidence for the exact directional pair carries authority. */
+export function compareReviewerCapability(
+  author: CapabilityModelIdentity,
+  reviewer: CapabilityModelIdentity,
+  current: CapabilityRevision,
+  records: readonly CapabilityPairRecord[],
+): CapabilityComparison {

+export function scoreCapabilityRun(fixture: CapabilityFixture, run: CapabilityRun): boolean {
+  const normalizedFindings = run.findings.map(finding => ({
+    severity: finding.severity,
+    message: finding.message.toLowerCase().replaceAll(/\s+/gu, ' '),
+  }));
+  const contains = (phrase: string, errorOnly: boolean): boolean =>
+    normalizedFindings.some(
+      finding =>
+        (!errorOnly || finding.severity === 'error') &&
+        finding.message.includes(phrase.toLowerCase().replaceAll(/\s+/gu, ' ')),
+    );
+  const groups = fixture.finding_groups;
+  return (
+    run.fixture_id === fixture.id &&
+    run.verdict === fixture.verdict &&
+    (run.verdict === 'request_changes') ===
+      normalizedFindings.some(finding => finding.severity === 'error') &&
+    fixture.required.every(finding => contains(finding, true)) &&
+    fixture.forbidden.every(finding => !contains(finding, false)) &&
+    (groups === undefined ||

+function invalidCorpusShape(
+  fixtures: readonly CapabilityFixture[],
+  runs: readonly CapabilityRun[],
+  floor: CapabilityFloor,
+): boolean {
+  return (
+    fixtures.length === 0 ||
+    new Set(fixtures.map(fixture => fixture.id)).size !== fixtures.length ||
+    !Number.isSafeInteger(floor.runs_per_fixture) ||
+    floor.runs_per_fixture < 1 ||
+    !Number.isSafeInteger(floor.minimum_fixture_passes) ||
+    floor.minimum_fixture_passes < 1 ||
+    floor.minimum_fixture_passes > floor.runs_per_fixture ||
+    floor.minimum_total_percent < 0 ||
+    floor.minimum_total_percent > 100 ||
+    runs.length !== fixtures.length * floor.runs_per_fixture
+  );
+}

+function passCounts(
+  fixtures: readonly CapabilityFixture[],
+  runs: readonly CapabilityRun[],
+  floor: CapabilityFloor,
+): ReadonlyMap<string, number> | undefined {
+  const byFixture = new Map(fixtures.map(fixture => [fixture.id, fixture]));
+  const seen = new Set<string>();
+  const passes = new Map(fixtures.map(fixture => [fixture.id, 0]));
+  for (const run of runs) {
+    const fixture = byFixture.get(run.fixture_id);
+    const key = `${run.fixture_id}:${run.run}`;
+    if (
+      fixture === undefined ||
+      !Number.isSafeInteger(run.run) ||
+      run.run < 1 ||
+      run.run > floor.runs_per_fixture ||
+      seen.has(key)
+    )
+      return undefined;
+    seen.add(key);
+    if (scoreCapabilityRun(fixture, run)) {
+      passes.set(fixture.id, (passes.get(fixture.id) ?? 0) + 1);
+    }
+  }
+  return passes;
+}
+
+function qualifiedPassCounts(
+  fixtures: readonly CapabilityFixture[],
+  runs: readonly CapabilityRun[],
+  floor: CapabilityFloor,
+): ReadonlyMap<string, number> | undefined {
+  if (invalidCorpusShape(fixtures, runs, floor)) return undefined;
+  const passes = passCounts(fixtures, runs, floor);
+  if (passes === undefined) return undefined;
+  let totalPasses = 0;
+  for (const count of passes.values()) {
+    if (count < floor.minimum_fixture_passes) return undefined;
+    totalPasses += count;
+  }
+  if (totalPasses * 100 < floor.minimum_total_percent * runs.length) return undefined;
+  return passes;
+}

+export function compareCapabilityRuns(
+  fixtures: readonly CapabilityFixture[],
+  author: readonly CapabilityRun[],
+  candidate: readonly CapabilityRun[],
+  floor: CapabilityFloor,
+): 'not_weaker' | 'weaker' | 'unknown' {
+  const authorPasses = qualifiedPassCounts(fixtures, author, floor);
+  const candidatePasses = qualifiedPassCounts(fixtures, candidate, floor);
+  if (authorPasses === undefined || candidatePasses === undefined) return 'unknown';
+  return [...authorPasses].every(([id, count]) => (candidatePasses.get(id) ?? 0) >= count)
+    ? 'not_weaker'
+    : 'weaker';
