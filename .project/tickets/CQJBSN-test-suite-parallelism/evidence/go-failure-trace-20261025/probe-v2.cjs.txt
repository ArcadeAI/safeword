require('./preload.cjs');
const cp = require('node:child_process');
const fs = require('node:fs');
const path = require('node:path');
const { promisify } = require('node:util');
const { syncBuiltinESMExports } = require('node:module');
const previous = cp.execFile;
function observed(...args) { return Reflect.apply(previous, this, args); }
for (const key of Reflect.ownKeys(previous)) {
  if (typeof key !== 'symbol') continue;
  const descriptor = Object.getOwnPropertyDescriptor(previous, key);
  if (key === promisify.custom) {
    const originalCustom = descriptor.value;
    descriptor.value = function (...args) {
      const result = Reflect.apply(originalCustom, this, args);
      const [command, parameters, options] = args;
      if (path.basename(command) !== 'go' || parameters?.[0] !== 'test' || !options?.cwd?.includes('safeword-go-context-')) return result;
      const start = Date.now();
      const file = `/Users/alex/.local/state/safeword/production-red-gate-B9M6FY/go-timeout-debug-20261025/go-test-${process.pid}-${start}.jsonl`;
      const record = (event, extra = {}) => fs.appendFileSync(file, JSON.stringify({ event, time: Date.now(), ...extra }) + '\n');
      record('go-test-start', { command, parameters, cwd: options.cwd, timeout: options.timeout ?? 0 });
      const child = result.child;
      let stdout = '', stderr = '';
      child.stdout?.on('data', chunk => { if (stdout.length < 65536) stdout += chunk.toString(); });
      child.stderr?.on('data', chunk => { if (stderr.length < 65536) stderr += chunk.toString(); });
      child.on('error', error => record('go-test-error', { message: error.message, code: error.code }));
      child.on('close', (code, signal) => record('go-test-close', { elapsed: Date.now() - start, code, signal, killed: child.killed, stdout, stderr }));
      return result;
    };
  }
  Object.defineProperty(observed, key, descriptor);
}
cp.execFile = observed;
syncBuiltinESMExports();
